// ===================== // AUTO FULLSCREEN // ===================== async function enterFullscreen() { const el = document.documentElement; try { if (!document.fullscreenElement && el.requestFullscreen) { await el.requestFullscreen(); return; } if (!document.fullscreenElement && el.webkitRequestFullscreen) { el.webkitRequestFullscreen(); return; } if (!document.fullscreenElement && el.msRequestFullscreen) { el.msRequestFullscreen(); return; } } catch (e) {} } // Try automatically when page loads window.addEventListener("load", () => { enterFullscreen(); }); // Try again when page becomes visible document.addEventListener("visibilitychange", () => { if (document.visibilityState === "visible") { enterFullscreen(); } }); // Try every 2 seconds until fullscreen succeeds setInterval(() => { if (!document.fullscreenElement) { enterFullscreen(); } }, 2000); // If browser requires user interaction, // first touch/click will enter fullscreen. ["click","touchstart","pointerdown"].forEach(event => { document.addEventListener(event, enterFullscreen, { once: true }); });
